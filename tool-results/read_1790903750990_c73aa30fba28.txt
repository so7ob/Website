     1→"use client";
     2→
     3→/**
     4→ * جدول إدارة المستخدمين: بحث مؤجل + تصفية (دور/حالة) + ترتيب + ترقيم صفحات،
     5→ * وإجراءات محمية بالصلاحيات: تعديل بيانات، تغيير دور، إيقاف/تفعيل، دعوة،
     6→ * إرسال استعادة كلمة المرور. العرض فقط إذنًا — القرار دائمًا في الخادم.
     7→ */
     8→import { useCallback, useEffect, useState } from "react";
     9→import Link from "next/link";
    10→import { toast } from "sonner";
    11→import {
    12→  Search,
    13→  MoreHorizontal,
    14→  Pencil,
    15→  ShieldCheck,
    16→  Ban,
    17→  CheckCircle2,
    18→  CircleX,
    19→  UserPlus,
    20→  KeyRound,
    21→  ArrowUpDown,
    22→  ArrowUp,
    23→  ArrowDown,
    24→  Users,
    25→  Loader2,
    26→  RotateCcw,
    27→  Eye,
    28→} from "lucide-react";
    29→import { Button } from "@/components/ui/button";
    30→import { Input } from "@/components/ui/input";
    31→import { Label } from "@/components/ui/label";
    32→import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from "@/components/ui/select";
    33→import { Dialog, DialogContent, DialogFooter, DialogHeader, DialogTitle } from "@/components/ui/dialog";
    34→import {
    35→  AlertDialog,
    36→  AlertDialogAction,
    37→  AlertDialogCancel,
    38→  AlertDialogContent,
    39→  AlertDialogDescription,
    40→  AlertDialogFooter,
    41→  AlertDialogHeader,
    42→  AlertDialogTitle,
    43→} from "@/components/ui/alert-dialog";
    44→import {
    45→  DropdownMenu,
    46→  DropdownMenuContent,
    47→  DropdownMenuItem,
    48→  DropdownMenuLabel,
    49→  DropdownMenuSeparator,
    50→  DropdownMenuTrigger,
    51→} from "@/components/ui/dropdown-menu";
    52→import { Table, TableBody, TableCell, TableHead, TableHeader, TableRow } from "@/components/ui/table";
    53→import { Skeleton } from "@/components/ui/skeleton";
    54→import { getPortalContent } from "@/content/portal";
    55→import { can } from "@/lib/auth/permissions";
    56→import { SYSTEM_ROLES } from "@/lib/auth/permissions";
    57→import type { Locale } from "@/lib/i18n";
    58→import { RoleBadge, UserStatusBadge } from "@/components/admin/badges";
    59→import { AdminPagination } from "@/components/admin/pagination";
    60→import { EmptyState } from "@/components/admin/empty-state";
    61→import { useDebounced } from "@/components/admin/use-debounced";
    62→import { apiGet, apiSend, ApiError, apiErrorMessage, buildQuery, fmtDate, fmtRelative, totalPages } from "@/components/admin/helpers";
    63→import type { AdminUser, Me, UsersResponse } from "../types";
    64→import { cn } from "@/lib/utils";
    65→
    66→type SortCol = "createdAt" | "lastLoginAt" | "name" | "email" | "roleKey";
    67→
    68→interface UsersClientProps {
    69→  me: Me;
    70→  locale: Locale;
    71→  initialQ: string;
    72→}
    73→
    74→export function UsersClient({ me, locale, initialQ }: UsersClientProps) {
    75→  const t = getPortalContent(locale);
    76→  const tu = t.admin.users;
    77→
    78→  const [q, setQ] = useState(initialQ);
    79→  const debouncedQ = useDebounced(q);
    80→  const [role, setRole] = useState("all");
    81→  const [status, setStatus] = useState("all");
    82→  const [sort, setSort] = useState<SortCol>("createdAt");
    83→  const [dir, setDir] = useState<"asc" | "desc">("desc");
    84→  const [page, setPage] = useState(1);
    85→  const [reloadToken, setReloadToken] = useState(0);
    86→
    87→  const [data, setData] = useState<UsersResponse | null>(null);
    88→  const [loading, setLoading] = useState(true);
    89→  const [error, setError] = useState<string | null>(null);
    90→
    91→  // ——— الجلب ———
    92→  const load = useCallback(
    93→    async (signal: AbortSignal) => {
    94→      setLoading(true);
    95→      setError(null);
    96→      try {
    97→        const query = buildQuery({
    98→          q: debouncedQ,
    99→          role: role !== "all" ? role : "",
   100→          status: status !== "all" ? status : "",
   101→          sort,
   102→          dir,
   103→          page,
   104→        });
   105→        const res = await apiGet<UsersResponse>(`/api/admin/users${query}`);
   106→        if (!signal.aborted) setData(res);
   107→      } catch (err) {
   108→        if (!signal.aborted && err instanceof ApiError) setError(apiErrorMessage(err, t.auth.errors));
   109→      } finally {
   110→        if (!signal.aborted) setLoading(false);
   111→      }
   112→    },
   113→    [debouncedQ, role, status, sort, dir, page, t.auth.errors]
   114→  );
   115→
   116→  useEffect(() => {
   117→    const controller = new AbortController();
   118→    load(controller.signal);
   119→    return () => controller.abort();
   120→  }, [load, reloadToken]);
   121→
   122→  const reload = () => setReloadToken((v) => v + 1);
   123→
   124→  // ——— الحواريات ———
   125→  const [inviteOpen, setInviteOpen] = useState(false);
   126→  const [inviteEmail, setInviteEmail] = useState("");
   127→  const [inviteRole, setInviteRole] = useState("client");
   128→  const [inviteSending, setInviteSending] = useState(false);
   129→  const [inviteUrl, setInviteUrl] = useState<string | null>(null);
   130→
   131→  const [editTarget, setEditTarget] = useState<AdminUser | null>(null);
   132→  const [editName, setEditName] = useState("");
   133→  const [editPhone, setEditPhone] = useState("");
   134→  const [editCompany, setEditCompany] = useState("");
   135→  const [editSaving, setEditSaving] = useState(false);
   136→
   137→  const [roleTarget, setRoleTarget] = useState<AdminUser | null>(null);
   138→  const [roleValue, setRoleValue] = useState("client");
   139→  const [roleSaving, setRoleSaving] = useState(false);
   140→
   141→  const [confirmTarget, setConfirmTarget] = useState<AdminUser | null>(null); // إيقاف/تفعيل
   142→
   143→  const mayEdit = can(me, "users.update");
   144→  const mayRoles = can(me, "users.roles");
   145→  const maySuspend = can(me, "users.suspend");
   146→  const mayInvite = can(me, "users.create");
   147→
   148→  // ——— التعديلات ———
   149→  const patchUser = async (id: string, body: Record<string, unknown>, successMsg: string) => {
   150→    try {
   151→      await apiSend(`/api/admin/users/${id}`, "PATCH", body);
   152→      toast.success(successMsg);
   153→      reload();
   154→    } catch (err) {
   155→      if (err instanceof ApiError && err.code === "last_admin") {
   156→        toast.error(tu.lastAdminError);
   157→      } else {
   158→        toast.error(apiErrorMessage(err, t.auth.errors));
   159→      }
   160→    }
   161→  };
   162→
   163→  const sendInvite = async () => {
   164→    const email = inviteEmail.trim().toLowerCase();
   165→    if (!/^[^\s@]+@[^\s@]+\.[^\s@]{2,}$/.test(email)) {
   166→      toast.error(t.auth.errors.emailInvalid);
   167→      return;
   168→    }
   169→    setInviteSending(true);
   170→    try {
   171→      const res = await apiSend<{ ok: boolean; emailStatus: string; devInviteUrl?: string }>("/api/admin/users", "POST", {
   172→        email,
   173→        roleKey: inviteRole,
   174→      });
   175→      toast.success(tu.inviteSent);
   176→      setInviteUrl(res.devInviteUrl ?? null);
   177→      setInviteEmail("");
   178→    } catch (err) {
   179→      toast.error(apiErrorMessage(err, t.auth.errors));
   180→    } finally {
   181→      setInviteSending(false);
   182→    }
   183→  };
   184→
   185→  const saveEdit = async () => {
   186→    if (!editTarget) return;
   187→    setEditSaving(true);
   188→    try {
   189→      await patchUser(editTarget.id, { name: editName, phone: editPhone, company: editCompany }, tu.saved);
   190→    } finally {
   191→      setEditSaving(false);
   192→      setEditTarget(null);
   193→    }
   194→  };
   195→
   196→  const saveRole = async () => {
   197→    if (!roleTarget) return;
   198→    setRoleSaving(true);
   199→    try {
   200→      await patchUser(roleTarget.id, { roleKey: roleValue }, tu.userUpdated);
   201→    } finally {
   202→      setRoleSaving(false);
   203→      setRoleTarget(null);
   204→    }
   205→  };
   206→
   207→  const applyStatus = async () => {
   208→    if (!confirmTarget) return;
   209→    const next = confirmTarget.status === "suspended" ? "active" : "suspended";
   210→    await patchUser(confirmTarget.id, { status: next }, tu.userUpdated);
   211→    setConfirmTarget(null);
   212→  };
   213→
   214→  const sendReset = async (user: AdminUser) => {
   215→    try {
   216→      await apiSend("/api/auth/forgot-password", "POST", { email: user.email });
   217→      toast.success(tu.resetSent);
   218→    } catch (err) {
   219→      toast.error(apiErrorMessage(err, t.auth.errors));
   220→    }
   221→  };
   222→
   223→  const onSort = (col: SortCol) => {
   224→    if (sort === col) {
   225→      setDir((d) => (d === "asc" ? "desc" : "asc"));
   226→    } else {
   227→      setSort(col);
   228→      setDir(col === "name" || col === "email" ? "asc" : "desc");
   229→    }
   230→    setPage(1);
   231→  };
   232→
   233→  const users = data?.users ?? [];
   234→  const statusLabel = (s: string) =>
   235→    s === "active" ? tu.statusActive : s === "pending_verification" ? tu.statusPending : tu.statusSuspended;
   236→
   237→  return (
   238→    <div className="space-y-5">
   239→      <div className="flex flex-wrap items-end justify-between gap-3">
   240→        <div>
   241→          <h1 className="text-2xl font-bold text-navy">{tu.title}</h1>
   242→          <p className="mt-1 text-sm text-muted-foreground">{tu.subtitle}</p>
   243→        </div>
   244→        {mayInvite ? (
   245→          <Button
   246→            onClick={() => {
   247→              setInviteUrl(null);
   248→              setInviteOpen(true);
   249→            }}
   250→            className="min-h-11 rounded-full"
   251→          >
   252→            <UserPlus className="size-4" aria-hidden="true" />
   253→            {tu.invite}
   254→          </Button>
   255→        ) : null}
   256→      </div>
   257→
   258→      {/* أدوات التصفية */}
   259→      <div className="flex flex-wrap items-center gap-3">
   260→        <div className="relative min-w-56 flex-1 sm:max-w-xs">
   261→          <Search className="absolute start-3 top-1/2 size-4 -translate-y-1/2 text-muted-foreground" aria-hidden="true" />
   262→          <Input
   263→            value={q}
   264→            onChange={(e) => {
   265→              setQ(e.target.value);
   266→              setPage(1);
   267→            }}
   268→            placeholder={tu.searchPlaceholder}
   269→            aria-label={tu.search}
   270→            className="min-h-11 ps-9"
   271→          />
   272→        </div>
   273→        <Select value={role} onValueChange={(v) => { setRole(v); setPage(1); }}>
   274→          <SelectTrigger aria-label={tu.role} className="min-h-11 w-40">
   275→            <SelectValue />
   276→          </SelectTrigger>
   277→          <SelectContent>
   278→            <SelectItem value="all">{tu.filterAll} — {tu.role}</SelectItem>
   279→            {SYSTEM_ROLES.map((r) => (
   280→              <SelectItem key={r.key} value={r.key}>
   281→                {locale === "en" ? r.nameEn : r.nameAr}
   282→              </SelectItem>
   283→            ))}
   284→          </SelectContent>
   285→        </Select>
   286→        <Select value={status} onValueChange={(v) => { setStatus(v); setPage(1); }}>
   287→          <SelectTrigger aria-label={tu.status} className="min-h-11 w-44">
   288→            <SelectValue />
   289→          </SelectTrigger>
   290→          <SelectContent>
   291→            <SelectItem value="all">{tu.filterAll} — {tu.status}</SelectItem>
   292→            <SelectItem value="active">{tu.statusActive}</SelectItem>
   293→            <SelectItem value="pending_verification">{tu.statusPending}</SelectItem>
   294→            <SelectItem value="suspended">{tu.statusSuspended}</SelectItem>
   295→          </SelectContent>
   296→        </Select>
   297→      </div>
   298→
   299→      {/* الجدول */}
   300→      <div className="overflow-hidden rounded-2xl border border-border bg-white">
   301→        <div className="overflow-x-auto">
   302→          <Table>
   303→            <TableHeader>
   304→              <TableRow className="bg-muted/50 hover:bg-muted/50">
   305→                <TableHead className="min-w-52">
   306→                  <SortHeader label={t.account.profile.name} col="name" sort={sort} dir={dir} onSort={onSort} />
   307→                </TableHead>
   308→                <TableHead className="min-w-32">
   309→                  <SortHeader label={tu.role} col="roleKey" sort={sort} dir={dir} onSort={onSort} />
   310→                </TableHead>
   311→                <TableHead className="min-w-28">{tu.status}</TableHead>
   312→                <TableHead className="w-16"><span className="sr-only">{tu.emailVerified}</span></TableHead>
   313→                <TableHead>
   314→                  <SortHeader label={tu.createdAt} col="createdAt" sort={sort} dir={dir} onSort={onSort} />
   315→                </TableHead>
   316→                <TableHead>
   317→                  <SortHeader label={tu.lastLogin} col="lastLoginAt" sort={sort} dir={dir} onSort={onSort} />
   318→                </TableHead>
   319→                <TableHead className="text-center">{tu.requests}</TableHead>
   320→                <TableHead className="w-14"><span className="sr-only">{tu.actions}</span></TableHead>
   321→              </TableRow>
   322→            </TableHeader>
   323→            <TableBody>
   324→              {loading && !data ? (
   325→                Array.from({ length: 6 }).map((_, i) => (
   326→                  <TableRow key={`sk-${i}`}>
   327→                    {Array.from({ length: 8 }).map((__, j) => (
   328→                      <TableCell key={j}>
   329→                        <Skeleton className="h-5 w-full" />
   330→                      </TableCell>
   331→                    ))}
   332→                  </TableRow>
   333→                ))
   334→              ) : users.length === 0 ? (
   335→                <TableRow>
   336→                  <TableCell colSpan={8} className="p-0">
   337→                    <EmptyState icon={Users} title={t.admin.dashboard.noData} />
   338→                  </TableCell>
   339→                </TableRow>
   340→              ) : (
   341→                users.map((user) => (
   342→                  <TableRow key={user.id} className="transition-colors hover:bg-muted/40">
   343→                    <TableCell>
   344→                      <div className="min-w-0">
   345→                        <Link
   346→                          href={`/${locale}/admin/users/${user.id}`}
   347→                          className="block truncate text-sm font-semibold text-navy transition-colors hover:text-brand"
   348→                        >
   349→                          {user.name}
   350→                        </Link>
   351→                        <p className="truncate text-xs text-muted-foreground ltr-isolate">{user.email}</p>
   352→                      </div>
   353→                    </TableCell>
   354→                    <TableCell><RoleBadge roleKey={user.roleKey} locale={locale} /></TableCell>
   355→                    <TableCell><UserStatusBadge status={user.status} label={statusLabel(user.status)} /></TableCell>
   356→                    <TableCell>
   357→                      {user.emailVerified ? (
   358→                        <span className="inline-flex items-center gap-1 rounded-full bg-emerald-100 px-2 py-0.5 text-[11px] font-medium text-emerald-800">
   359→                          <CheckCircle2 className="size-3" aria-hidden="true" />
   360→                          <span className="sr-only">{tu.emailVerified}</span>
   361→                        </span>
   362→                      ) : (
   363→                        <span className="inline-flex items-center gap-1 rounded-full bg-muted px-2 py-0.5 text-[11px] font-medium text-muted-foreground">
   364→                          <CircleX className="size-3" aria-hidden="true" />
   365→                          <span className="sr-only">{tu.notVerified}</span>
   366→                        </span>
   367→                      )}
   368→                    </TableCell>
   369→                    <TableCell className="whitespace-nowrap text-sm text-muted-foreground">{fmtDate(user.createdAt, locale)}</TableCell>
   370→                    <TableCell className="whitespace-nowrap text-sm text-muted-foreground">
   371→                      {user.lastLoginAt ? fmtRelative(user.lastLoginAt, locale) : tu.never}
   372→                    </TableCell>
   373→                    <TableCell className="text-center text-sm font-medium tabular-nums text-navy">{user.requestsCount}</TableCell>
   374→                    <TableCell>
   375→                      <DropdownMenu>
   376→                        <DropdownMenuTrigger asChild>
   377→                          <Button variant="ghost" size="icon" className="size-10" aria-label={tu.actions}>
   378→                            <MoreHorizontal className="size-4" aria-hidden="true" />
   379→                          </Button>
   380→                        </DropdownMenuTrigger>
   381→                        <DropdownMenuContent align="end" className="w-56">
   382→                          <DropdownMenuLabel className="truncate">{user.name}</DropdownMenuLabel>
   383→                          <DropdownMenuSeparator />
   384→                          <DropdownMenuItem asChild>
   385→                            <Link href={`/${locale}/admin/users/${user.id}`}>
   386→                              <Eye className="size-4" aria-hidden="true" />
   387→                              {tu.viewProfile}
   388→                            </Link>
   389→                          </DropdownMenuItem>
   390→                          {mayEdit ? (
   391→                            <DropdownMenuItem
   392→                              onClick={() => {
   393→                                setEditTarget(user);
   394→                                setEditName(user.name);
   395→                                setEditPhone(user.phone ?? "");
   396→                                setEditCompany(user.company ?? "");
   397→                              }}
   398→                            >
   399→                              <Pencil className="size-4" aria-hidden="true" />
   400→                              {tu.edit}
   401→                            </DropdownMenuItem>
   402→                          ) : null}
   403→                          {mayRoles ? (
   404→                            <DropdownMenuItem
   405→                              onClick={() => {
   406→                                setRoleTarget(user);
   407→                                setRoleValue(user.roleKey);
   408→                              }}
   409→                            >
   410→                              <ShieldCheck className="size-4" aria-hidden="true" />
   411→                              {tu.changeRole}
   412→                            </DropdownMenuItem>
   413→                          ) : null}
   414→                          <DropdownMenuItem onClick={() => sendReset(user)}>
   415→                            <KeyRound className="size-4" aria-hidden="true" />
   416→                            {tu.sendReset}
   417→                          </DropdownMenuItem>
   418→                          {maySuspend ? (
   419→                            <>
   420→                              <DropdownMenuSeparator />
   421→                              <DropdownMenuItem
   422→                                disabled={user.id === me.id}
   423→                                onClick={() => setConfirmTarget(user)}
   424→                                className={user.status === "suspended" ? "" : "text-destructive focus:text-destructive"}
   425→                              >
   426→                                {user.status === "suspended" ? (
   427→                                  <CheckCircle2 className="size-4" aria-hidden="true" />
   428→                                ) : (
   429→                                  <Ban className="size-4" aria-hidden="true" />
   430→                                )}
   431→                                {user.status === "suspended" ? tu.activate : tu.suspend}
   432→                              </DropdownMenuItem>
   433→                            </>
   434→                          ) : null}
   435→                        </DropdownMenuContent>
   436→                      </DropdownMenu>
   437→                    </TableCell>
   438→                  </TableRow>
   439→                ))
   440→              )}
   441→            </TableBody>
   442→          </Table>
   443→        </div>
   444→      </div>
   445→
   446→      {error ? (
   447→        <div className="flex items-center justify-between gap-3 rounded-2xl border border-destructive/30 bg-destructive/5 px-4 py-3">
   448→          <p className="text-sm text-destructive">{error}</p>
   449→          <Button variant="outline" size="icon" onClick={reload} className="size-10 shrink-0" aria-label={tu.search}>
   450→            <RotateCcw className="size-4" aria-hidden="true" />
   451→          </Button>
   452→        </div>
   453→      ) : null}
   454→
   455→      {data ? (
   456→        <AdminPagination
   457→          page={page}
   458→          total={data.total}
   459→          pageSize={data.pageSize}
   460→          locale={locale}
   461→          onPage={setPage}
   462→        />
   463→      ) : null}
   464→
   465→      {loading && data ? (
   466→        <Loader2 className="size-4 animate-spin text-muted-foreground" aria-hidden="true" />
   467→      ) : null}
   468→
   469→      {/* حوارية الدعوة */}
   470→      <Dialog open={inviteOpen} onOpenChange={(open) => { setInviteOpen(open); if (!open) setInviteUrl(null); }}>
   471→        <DialogContent className="sm:max-w-md">
   472→          <DialogHeader>
   473→            <DialogTitle className="text-navy">{tu.inviteTitle}</DialogTitle>
   474→          </DialogHeader>
   475→          <div className="space-y-4">
   476→            <div className="space-y-2">
   477→              <Label htmlFor="invite-email">{tu.inviteEmail}</Label>
   478→              <Input
   479→                id="invite-email"
   480→                type="email"
   481→                dir="ltr"
   482→                value={inviteEmail}
   483→                onChange={(e) => setInviteEmail(e.target.value)}
   484→                className="min-h-11"
   485→                autoComplete="off"
   486→              />
   487→            </div>
   488→            <div className="space-y-2">
   489→              <Label>{tu.inviteRole}</Label>
   490→              <Select value={inviteRole} onValueChange={setInviteRole}>
   491→                <SelectTrigger className="min-h-11">
   492→                  <SelectValue />
   493→                </SelectTrigger>
   494→                <SelectContent>
   495→                  {SYSTEM_ROLES.map((r) => (
   496→                    <SelectItem key={r.key} value={r.key}>
   497→                      {locale === "en" ? r.nameEn : r.nameAr}
   498→                    </SelectItem>
   499→                  ))}
   500→                </SelectContent>
   501→              </Select>
   502→            </div>
   503→            {inviteUrl ? (
   504→              <div className="rounded-xl border border-dashed border-brand bg-accent/50 p-3">
   505→                <p className="text-xs font-medium text-brand-strong">{t.auth.devOutboxHint}</p>
   506→                <div className="mt-2 flex items-center gap-2">
   507→                  <code className="min-w-0 flex-1 truncate rounded-md bg-white px-2 py-1.5 text-xs ltr-isolate" dir="ltr">
   508→                    {inviteUrl}
   509→                  </code>
   510→                  <Button
   511→                    type="button"
   512→                    variant="outline"
   513→                    size="sm"
   514→                    className="min-h-9"
   515→                    onClick={() => {
   516→                      void navigator.clipboard?.writeText(inviteUrl);
   517→                    }}
   518→                  >
   519→                    {t.admin.media.copyUrl}
   520→                  </Button>
   521→                </div>
   522→              </div>
   523→            ) : null}
   524→          </div>
   525→          <DialogFooter className="gap-2">
   526→            <Button variant="outline" onClick={() => setInviteOpen(false)} className="min-h-11 rounded-full">
   527→              {tu.cancel}
   528→            </Button>
   529→            <Button onClick={sendInvite} disabled={inviteSending} className="min-h-11 rounded-full">
   530→              {inviteSending ? <Loader2 className="size-4 animate-spin" aria-hidden="true" /> : null}
   531→              {tu.inviteSend}
   532→            </Button>
   533→          </DialogFooter>
   534→        </DialogContent>
   535→      </Dialog>
   536→
   537→      {/* حوارية تعديل البيانات */}
   538→      <Dialog open={editTarget !== null} onOpenChange={(open) => { if (!open) setEditTarget(null); }}>
   539→        <DialogContent className="sm:max-w-md">
   540→          <DialogHeader>
   541→            <DialogTitle className="text-navy">{tu.edit} — {editTarget?.name}</DialogTitle>
   542→          </DialogHeader>
   543→          <div className="space-y-4">
   544→            <div className="space-y-2">
   545→              <Label htmlFor="edit-name">{t.account.profile.name}</Label>
   546→              <Input id="edit-name" value={editName} onChange={(e) => setEditName(e.target.value)} className="min-h-11" maxLength={100} />
   547→            </div>
   548→            <div className="space-y-2">
   549→              <Label htmlFor="edit-phone">{t.account.profile.phone}</Label>
   550→              <Input id="edit-phone" dir="ltr" value={editPhone} onChange={(e) => setEditPhone(e.target.value)} className="min-h-11" maxLength={20} />
   551→            </div>
   552→            <div className="space-y-2">
   553→              <Label htmlFor="edit-company">{t.account.profile.company}</Label>
   554→              <Input id="edit-company" value={editCompany} onChange={(e) => setEditCompany(e.target.value)} className="min-h-11" maxLength={120} />
   555→            </div>
   556→          </div>
   557→          <DialogFooter className="gap-2">
   558→            <Button variant="outline" onClick={() => setEditTarget(null)} className="min-h-11 rounded-full">
   559→              {tu.cancel}
   560→            </Button>
   561→            <Button onClick={saveEdit} disabled={editSaving} className="min-h-11 rounded-full">
   562→              {editSaving ? <Loader2 className="size-4 animate-spin" aria-hidden="true" /> : null}
   563→              {tu.save}
   564→            </Button>
   565→          </DialogFooter>
   566→        </DialogContent>
   567→      </Dialog>
   568→
   569→      {/* حوارية تغيير الدور */}
   570→      <Dialog open={roleTarget !== null} onOpenChange={(open) => { if (!open) setRoleTarget(null); }}>
   571→        <DialogContent className="sm:max-w-md">
   572→          <DialogHeader>
   573→            <DialogTitle className="text-navy">{tu.changeRole} — {roleTarget?.name}</DialogTitle>
   574→          </DialogHeader>
   575→          <div className="space-y-2">
   576→            <Label>{tu.role}</Label>
   577→            <Select value={roleValue} onValueChange={setRoleValue}>
   578→              <SelectTrigger className="min-h-11">
   579→                <SelectValue />
   580→              </SelectTrigger>
   581→              <SelectContent>
   582→                {SYSTEM_ROLES.map((r) => (
   583→                  <SelectItem key={r.key} value={r.key}>
   584→                    {locale === "en" ? r.nameEn : r.nameAr}
   585→                  </SelectItem>
   586→                ))}
   587→              </SelectContent>
   588→            </Select>
   589→            <p className="text-xs text-muted-foreground">{tu.confirmLastAdmin}</p>
   590→          </div>
   591→          <DialogFooter className="gap-2">
   592→            <Button variant="outline" onClick={() => setRoleTarget(null)} className="min-h-11 rounded-full">
   593→              {tu.cancel}
   594→            </Button>
   595→            <Button onClick={saveRole} disabled={roleSaving || roleValue === roleTarget?.roleKey} className="min-h-11 rounded-full">
   596→              {roleSaving ? <Loader2 className="size-4 animate-spin" aria-hidden="true" /> : null}
   597→              {tu.save}
   598→            </Button>
   599→          </DialogFooter>
   600→        </DialogContent>
   601→      </Dialog>
   602→
   603→      {/* تأكيد الإيقاف/التفعيل */}
   604→      <AlertDialog open={confirmTarget !== null} onOpenChange={(open) => { if (!open) setConfirmTarget(null); }}>
   605→        <AlertDialogContent>
   606→          <AlertDialogHeader>
   607→            <AlertDialogTitle className="text-navy">
   608→              {confirmTarget?.status === "suspended" ? tu.activate : tu.suspend} — {confirmTarget?.name}
   609→            </AlertDialogTitle>
   610→            <AlertDialogDescription>
   611→              {confirmTarget?.status === "suspended" ? tu.confirmActivate : tu.confirmSuspend}
   612→            </AlertDialogDescription>
   613→          </AlertDialogHeader>
   614→          <AlertDialogFooter>
   615→            <AlertDialogCancel className="min-h-11 rounded-full">{tu.cancel}</AlertDialogCancel>
   616→            <AlertDialogAction onClick={applyStatus} className="min-h-11 rounded-full">
   617→              {confirmTarget?.status === "suspended" ? tu.activate : tu.suspend}
   618→            </AlertDialogAction>
   619→          </AlertDialogFooter>
   620→        </AlertDialogContent>
   621→      </AlertDialog>
   622→    </div>
   623→  );
   624→}

   626→/** رأس عمود قابل للترتيب */
   627→function SortHeader({
   628→  label,
   629→  col,
   630→  sort,
   631→  dir,
   632→  onSort,
   633→}: {
   634→  label: string;
   635→  col: SortCol;
   636→  sort: SortCol;
   637→  dir: "asc" | "desc";
   638→  onSort: (col: SortCol) => void;
   639→}) {
   640→  const active = sort === col;
   641→  const Icon = !active ? ArrowUpDown : dir === "asc" ? ArrowUp : ArrowDown;
   642→  return (
   643→    <button
   644→      type="button"
   645→      onClick={() => onSort(col)}
   646→      className={cn(
   647→        "inline-flex items-center gap-1 rounded-md text-xs font-medium transition-colors hover:text-navy",
   648→        active ? "text-navy" : "text-muted-foreground"
   649→      )}
   650→    >
   651→      {label}
   652→      <Icon className="size-3.5" aria-hidden="true" />
   653→    </button>
   654→  );
   655→}
