     7→ */
     8→import { useCallback, useEffect, useMemo, useRef, useState } from "react";
     9→import Link from "next/link";
    10→import { toast } from "sonner";
    11→import {
    12→  ArrowLeft,
    13→  ArrowRight,
    14→  Eye,
    15→  EyeOff,
    16→  UserPlus,
    17→  Paperclip,
    18→  Download,
    19→  Archive,
    20→  ArchiveRestore,
    21→  Inbox,
    22→  Loader2,
    23→  RotateCcw,
    24→  History,
    25→  Building2,
    26→} from "lucide-react";
    27→import { Button } from "@/components/ui/button";
    28→import { Input } from "@/components/ui/input";
    29→import { Label } from "@/components/ui/label";
    30→import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from "@/components/ui/select";
    31→import { Skeleton } from "@/components/ui/skeleton";
    32→import { getPortalContent } from "@/content/portal";
    33→import { ar as siteAr } from "@/content/ar";
    34→import { en as siteEn } from "@/content/en";
    35→import { can } from "@/lib/auth/permissions";
    36→import { localeMeta, type Locale } from "@/lib/i18n";
    37→import { StatusBadge, PriorityBadge, UserStatusBadge } from "@/components/admin/badges";
    38→import { EmptyState } from "@/components/admin/empty-state";
    39→import { MessageBubble, ReplyComposer } from "@/components/admin/conversation";
    40→import {
    41→  apiGet,
    42→  apiSend,

    44→  ApiError,

    46→  buildQuery,
    47→  fmtDate,
    48→  fmtDateTime,
    49→  formatBytes,
    50→} from "@/components/admin/helpers";

    53→
    54→interface RequestDetailClientProps {
    55→  me: Me;
    56→  locale: Locale;
    57→  requestId: string;
    58→}
    59→
    60→const STATUS_KEYS = ["new", "in_review", "awaiting_info", "in_progress", "responded", "closed", "cancelled"];
    61→const PRIORITY_KEYS = ["low", "normal", "high", "urgent"];
    62→
    63→export function RequestDetailClient({ me, locale, requestId }: RequestDetailClientProps) {
    64→  const t = getPortalContent(locale);
    65→  const tr = t.admin.requests;
    66→  const td = t.account.detail;
    67→  const siteForm = locale === "en" ? siteEn.form : siteAr.form;
    68→
    69→  const [detail, setDetail] = useState<RequestDetail | null>(null);
    70→  const [staffOptions, setStaffOptions] = useState<StaffOption[]>([]);
    71→  const [loading, setLoading] = useState(true);
    72→  const [notFound, setNotFound] = useState(false);
    73→  const [error, setError] = useState<string | null>(null);
    74→  const [reloadToken, setReloadToken] = useState(0);
    75→
    76→  const [clientView, setClientView] = useState(false);
    77→  const [sending, setSending] = useState(false);
    78→
    79→  // إدارة الحالة/الأولوية/التعيين
    80→  const [statusTarget, setStatusTarget] = useState("");
    81→  const [statusNote, setStatusNote] = useState("");
    82→  const [closeReason, setCloseReason] = useState("");
    83→  const [applying, setApplying] = useState(false);
    84→  const [busy, setBusy] = useState(false);
    85→
    86→  const busyRef = useRef(false);
    87→
    88→  const load = useCallback(
    89→    async (signal: AbortSignal, quiet = false) => {
    90→      if (!quiet) setLoading(true);
    91→      try {
    92→        const res = await apiGet<RequestDetailResponse>(`/api/admin/requests/${requestId}`);
    93→        if (!signal.aborted) {
    94→          setDetail(res.request);
    95→          setNotFound(false);
    96→          setError(null);
    97→          setStatusTarget((prev) => prev || res.request.status);
    98→        }
    99→      } catch (err) {
   100→        if (signal.aborted) return;
   101→        if (err instanceof ApiError && (err.code === "not_found" || err.status === 404)) {
   102→          setNotFound(true);
   103→        } else if (err instanceof ApiError) {
   104→          setError(apiErrorMessage(err, t.auth.errors));
   105→        }
   106→      } finally {
   107→        if (!signal.aborted) setLoading(false);
   108→      }
   109→    },
   110→    [requestId, t.auth.errors]
   111→  );
   112→
   113→  useEffect(() => {
   114→    const controller = new AbortController();
   115→    void load(controller.signal);
   116→    return () => controller.abort();
   117→  }, [load, reloadToken]);
   118→
   119→  // قائمة الطاقم القابلة للتعيين (تأتي مع استجابة قائمة الطلبات)
   120→  useEffect(() => {
   121→    const controller = new AbortController();
   122→    apiGet<RequestsResponse>("/api/admin/requests?page=1")
   123→      .then((res) => {
   124→        if (!controller.signal.aborted) setStaffOptions(res.staff);
   125→      })
   126→      .catch(() => {
   127→        // قائمة الطاقم تحسّن فقط — الفشل غير حرج
   128→      });
   129→    return () => controller.abort();
   130→  }, []);
   131→
   132→  // تحديث دوري كل 20 ثانية (يتوقف أثناء الإرسال أو حين تكون الصفحة مخفية)
   133→  useEffect(() => {
   134→    const timer = setInterval(() => {
   135→      if (busyRef.current || document.hidden) return;
   136→      const controller = new AbortController();
   137→      void load(controller.signal, true);
   138→    }, 20000);
   139→    return () => clearInterval(timer);
   140→  }, [load]);
   141→
   142→  const reload = () => setReloadToken((v) => v + 1);
   143→
   144→  // ——— الأذونات ———
   145→  const mayReply = can(me, "requests.reply");
   146→  const mayNote = can(me, "requests.internal_notes");
   147→  const mayAssign = can(me, "requests.assign");
   148→  const mayStatus = can(me, "requests.status");
   149→  const mayArchive = can(me, "requests.archive");
   150→
   151→  // ——— الإجراءات ———
   152→  const patch = async (body: Record<string, unknown>, successMsg?: string) => {
   153→    setBusy(true);
   154→    busyRef.current = true;
   155→    try {
   156→      await apiSend(`/api/admin/requests/${requestId}`, "PATCH", body);
   157→      if (successMsg) toast.success(successMsg);
   158→      reload();
   159→    } catch (err) {
   160→      if (err instanceof ApiError && err.code === "invalid_transition") {
   161→        toast.error(t.auth.errors.generic);
   162→      } else {
   163→        toast.error(apiErrorMessage(err, t.auth.errors));
   164→      }
   165→    } finally {
   166→      setBusy(false);
   167→      busyRef.current = false;
   168→    }
   169→  };
   170→
   171→  const sendMessage = async (kind: "message" | "internal_note", body: string) => {
   172→    setSending(true);
   173→    busyRef.current = true;
   174→    try {
   175→      const res = await apiSend<{ ok: boolean; message: MessageRow }>(
   176→        `/api/account/requests/${requestId}/messages`,
   177→        "POST",
   178→        { body, kind }
   179→      );
   180→      // إلحاق متفائل ثم جلب هادئ لتوحيد البيانات
   181→      setDetail((prev) =>
   182→        prev
   183→          ? {
   184→              ...prev,
   185→              messages: [
   186→                ...prev.messages,
   187→                {
   188→                  ...res.message,
   189→                  author: { id: me.id, name: me.name, roleKey: me.roleKey },
   190→                },
   191→              ],
   192→            }
   193→          : prev
   194→      );
   195→      toast.success(kind === "internal_note" ? tr.sentNote : tr.sentReply);
   196→      const controller = new AbortController();
   197→      void load(controller.signal, true);
   198→    } catch (err) {
   199→      toast.error(apiErrorMessage(err, t.auth.errors));
   200→    } finally {
   201→      setSending(false);
   202→      busyRef.current = false;
   203→    }
   204→  };
   205→
   206→  const attachFile = async (file: File) => {
   207→    busyRef.current = true;
   208→    try {
   209→      const form = new FormData();
   210→      form.append("file", file);
   211→      form.append("requestId", requestId);
   212→      await apiUpload("/api/attachments", form);
   213→      toast.success(t.admin.media.uploadedOk);
   214→      reload();
   215→    } catch (err) {
   216→      if (err instanceof ApiError && (err.code === "too_large" || err.code === "file_too_large")) {
   217→        toast.error(td.fileTooLarge);
   218→      } else if (err instanceof ApiError && (err.code === "type_not_allowed" || err.code === "extension_mismatch")) {
   219→        toast.error(td.fileTypeInvalid);
   220→      } else {
   221→        toast.error(apiErrorMessage(err, t.auth.errors));
   222→      }
   223→    } finally {
   224→      busyRef.current = false;
   225→    }
   226→  };
   227→
   228→  const applyStatus = async () => {
   229→    if (!statusTarget) return;
   230→    const isClosure = statusTarget === "closed" || statusTarget === "cancelled";
   231→    if (isClosure && closeReason.trim().length === 0) {
   232→      toast.error(t.auth.errors.required);
   233→      return;
   234→    }
   235→    setApplying(true);
   236→    try {
   237→      await patch({ status: statusTarget, note: isClosure ? closeReason.trim() : statusNote.trim() || undefined }, tr.statuses[statusTarget]);
   238→      setStatusNote("");
   239→      setCloseReason("");
   240→    } finally {
   241→      setApplying(false);
   242→    }
   243→  };
   244→
   245→  const visibleMessages = useMemo(
   246→    () => (detail ? detail.messages.filter((m) => !(clientView && m.kind === "internal_note")) : []),
   247→    [detail, clientView]
   248→  );
   249→
   250→  // ——— الحالات ———
   251→  if (loading && !detail) {
   252→    return (
   253→      <div className="grid gap-6 lg:grid-cols-[minmax(0,1fr)_360px]">
   254→        <div className="space-y-4">
   255→          <Skeleton className="h-9 w-64 rounded-xl" />
   256→          <Skeleton className="h-96 rounded-2xl" />
   257→        </div>
   258→        <div className="space-y-4">
   259→          <Skeleton className="h-40 rounded-2xl" />
   260→          <Skeleton className="h-56 rounded-2xl" />
   261→          <Skeleton className="h-40 rounded-2xl" />
   262→        </div>
   263→      </div>
   264→    );
   265→  }
   266→
   267→  if (notFound) {
   268→    return <EmptyState icon={Inbox} title={tr.empty} body={tr.emptyBody} className="py-20" />;
   269→  }
   270→
   271→  if (!detail) {
   272→    return (
   273→      <div className="flex flex-col items-center gap-4 py-20">
   274→        <p className="text-sm text-destructive">{error ?? t.auth.errors.generic}</p>
   275→        <Button variant="outline" onClick={reload} className="min-h-11 rounded-full">
   276→          <RotateCcw className="size-4" aria-hidden="true" />
   277→        </Button>
   278→      </div>
   279→    );
   280→  }
   281→
   282→  const BackIcon = locale === "ar" ? ArrowRight : ArrowLeft;
   283→  const statusLabel = tr.statuses[detail.status] ?? detail.status;
   284→  const isClosureTarget = statusTarget === "closed" || statusTarget === "cancelled";
   285→  const contactPrefLabel =
   286→    detail.preferredContact === "email" ? siteForm.contactMethods.email
   287→    : detail.preferredContact === "phone" ? siteForm.contactMethods.phone
   288→    : siteForm.contactMethods.any;
   289→  const budgetLabel = siteForm.budgets[detail.budget as keyof typeof siteForm.budgets] ?? detail.budget;
   290→  const currencyLabel = detail.currency
   291→    ? (siteForm.currencies[detail.currency as keyof typeof siteForm.currencies] ?? detail.currency)
   292→    : null;
   293→  const timelineLabel = siteForm.timelines[detail.timeline as keyof typeof siteForm.timelines] ?? detail.timeline;
   294→
   295→  return (
   296→    <div className="space-y-5">
   297→      {/* الترويسة */}
   298→      <div className="flex flex-wrap items-center gap-3">
   299→        <Button asChild variant="ghost" size="sm" className="min-h-11 rounded-full">
   300→          <Link href={`/${locale}/admin/requests`}>
   301→            <BackIcon className="size-4" aria-hidden="true" />
   302→            <span className="sr-only">{tr.title}</span>
   303→            {tr.title}
   304→          </Link>
   305→        </Button>
   306→        <h1 className="font-mono text-xl font-bold text-navy ltr-isolate">{detail.refCode}</h1>
   307→        <StatusBadge status={detail.status} label={statusLabel} />
   308→        <PriorityBadge priority={detail.priority} label={tr.priorities[detail.priority] ?? detail.priority} />
   309→        {detail.archivedAt ? (
   310→          <span className="inline-flex items-center gap-1 rounded-full bg-muted px-2.5 py-0.5 text-xs font-medium text-muted-foreground">
   311→            <Archive className="size-3" aria-hidden="true" />
   312→            {tr.archived}
   313→          </span>
   314→        ) : null}
   315→        <Button
   316→          variant="outline"
   317→          size="sm"
   318→          onClick={() => setClientView((v) => !v)}
   319→          aria-pressed={clientView}
   320→          className="ms-auto min-h-11 rounded-full"
   321→        >
   322→          {clientView ? <EyeOff className="size-4" aria-hidden="true" /> : <Eye className="size-4" aria-hidden="true" />}
   323→          {tr.clientView}
   324→        </Button>
   325→      </div>
   326→
   327→      <div className="grid items-start gap-6 lg:grid-cols-[minmax(0,1fr)_360px]">
   328→        {/* المحادثة + الملحن */}
   329→        <div className="space-y-4">
   330→          <section aria-label={tr.conversation} className="rounded-2xl border border-border bg-muted/30 p-4 sm:p-5">
   331→            <h2 className="mb-3 text-sm font-semibold text-navy">
   332→              {tr.conversation}
   333→              <span className="ms-2 font-normal text-muted-foreground">
   334→                {detail.messages.length} {tr.messages}
   335→              </span>
   336→            </h2>
   337→
   338→            {visibleMessages.length === 0 && !detail.description ? (
   339→              <EmptyState icon={Inbox} title={td.noMessages} />
   340→            ) : (
   341→              <div className="flex max-h-[60vh] flex-col gap-1 overflow-y-auto pe-1">
   342→                {/* نص الطلب الأصلي كفقاعة افتتاحية من العميل */}
   343→                {detail.description ? (
   344→                  <MessageBubble
   345→                    locale={locale}
   346→                    message={{
   347→                      id: "opening",
   348→                      authorId: null,
   349→                      authorType: "client",
   350→                      kind: "message",
   351→                      body: detail.description,
   352→                      editedAt: null,
   353→                      createdAt: detail.createdAt,
   354→                      author: null,
   355→                    }}
   356→                    labels={{
   357→                      client: detail.name,
   358→                      staff: td.staff,
   359→                      system: td.system,
   360→                      internalHint: tr.internalNoteHint,
   361→                      statuses: tr.statuses,
   362→                    }}
   363→                  />
   364→                ) : null}
   365→                {visibleMessages.map((message) => (
   366→                  <MessageBubble
   367→                    key={message.id}
   368→                    message={message}
   369→                    locale={locale}
   370→                    labels={{
   371→                      client: tr.client,
   372→                      staff: td.staff,
   373→                      system: td.system,
   374→                      internalHint: tr.internalNoteHint,
   375→                      statuses: tr.statuses,
   376→                    }}
   377→                  />
   378→                ))}
   379→              </div>
   380→            )}
   381→          </section>
   382→
   383→          <ReplyComposer
   384→              canReply={mayReply}
   385→              canNote={!clientView && mayNote}
   386→              sending={sending}
   387→              onSend={sendMessage}
   388→              onAttach={clientView ? undefined : attachFile}
   389→              savedReplies={
   390→                mayReply && !clientView
   391→                  ? {
   392→                      trigger: t.admin.savedReplies.useReply,
   393→                      insert: t.admin.savedReplies.insert,
   394→                      empty: t.admin.savedReplies.empty,
   395→                    }
   396→                  : undefined
   397→              }
   398→              labels={{
   399→                reply: tr.reply,
   400→                internalNote: tr.internalNote,
   401→                internalHint: tr.internalNoteHint,
   402→                placeholder: td.replyPlaceholder,
   403→                send: td.send,
   404→                sending: td.sending,
   405→                attachFile: td.attachFile,
   406→              }}
   407→            />
   408→        </div>
   409→
   410→        {/* عمود المعلومات */}
   411→        <aside className="space-y-4">
   412→          {/* العميل */}
   413→          <section className="rounded-2xl border border-border bg-white p-4">
   414→            <h2 className="text-sm font-semibold text-navy">{tr.client}</h2>
   415→            <dl className="mt-3 space-y-2 text-sm">
   416→              <div className="flex items-start justify-between gap-3">
   417→                <dt className="text-muted-foreground">{t.account.profile.name}</dt>
   418→                <dd className="text-end font-medium text-navy">
   419→                  {detail.client?.name ?? detail.name}
   420→                  {detail.client?.status ? <UserStatusBadge
   421→                    status={detail.client.status}
   422→                    label={detail.client.status === "active" ? t.admin.users.statusActive : detail.client.status === "pending_verification" ? t.admin.users.statusPending : t.admin.users.statusSuspended}
   423→                    className="ms-2 align-top"
   424→                  /> : null}
   425→                </dd>
   426→              </div>
   427→              <div className="flex items-start justify-between gap-3">
   428→                <dt className="text-muted-foreground">{t.admin.settings.email}</dt>
   429→                <dd className="min-w-0 truncate text-end">
   430→                  <a href={`mailto:${detail.email}`} className="break-all text-brand transition-colors hover:text-brand-strong ltr-isolate">
   431→                    {detail.email}
   432→                  </a>
   433→                </dd>
   434→              </div>
   435→              {detail.phone ? (
   436→                <div className="flex items-start justify-between gap-3">
   437→                  <dt className="text-muted-foreground">{t.admin.settings.phone}</dt>
   438→                  <dd className="text-end ltr-isolate">{detail.phone}</dd>
   439→                </div>
   440→              ) : null}
   441→              {detail.company ? (
   442→                <div className="flex items-start justify-between gap-3">
   443→                  <dt className="text-muted-foreground">{t.account.profile.company}</dt>
   444→                  <dd className="flex items-center gap-1.5 text-end">
   445→                    <Building2 className="size-3.5 text-muted-foreground" aria-hidden="true" />
   446→                    {detail.company}
   447→                  </dd>
   448→                </div>
   449→              ) : null}
   450→            </dl>
   451→            {detail.clientId ? (
   452→              <Button asChild variant="outline" size="sm" className="mt-3 min-h-10 w-full rounded-xl">
   453→                <Link href={`/${locale}/admin/users${buildQuery({ q: detail.email })}`}>
   454→                  {t.admin.users.title}
   455→                </Link>
   456→              </Button>
   457→            ) : null}
   458→          </section>
   459→
   460→          {/* بيانات الطلب */}
   461→          <section className="rounded-2xl border border-border bg-white p-4">
   462→            <h2 className="text-sm font-semibold text-navy">{td.requestInfo}</h2>
   463→            <dl className="mt-3 space-y-2 text-sm">
   464→              <div className="flex items-start justify-between gap-3">
   465→                <dt className="text-muted-foreground">{tr.requestType}</dt>
   466→                <dd className="font-medium text-navy">
   467→                  {detail.requestType === "quote" ? tr.quote : tr.discussion}
   468→                </dd>
   469→              </div>
   470→              <div className="flex items-start justify-between gap-3">
   471→                <dt className="text-muted-foreground">{tr.filterService}</dt>
   472→                <dd className="text-end">{tr.services[detail.serviceType] ?? detail.serviceType}</dd>
   473→              </div>
   474→              <div className="flex items-start justify-between gap-3">
   475→                <dt className="text-muted-foreground">{tr.budget}</dt>
   476→                <dd className="text-end">{budgetLabel}{currencyLabel ? ` (${currencyLabel})` : ""}</dd>
   477→              </div>
   478→              <div className="flex items-start justify-between gap-3">
   479→                <dt className="text-muted-foreground">{tr.timelinePref}</dt>
   480→                <dd className="text-end">{timelineLabel}</dd>
   481→              </div>
   482→              <div className="flex items-start justify-between gap-3">
   483→                <dt className="text-muted-foreground">{tr.contactPref}</dt>
   484→                <dd className="text-end">{contactPrefLabel}</dd>
   485→              </div>
   486→              {detail.referenceUrl ? (
   487→                <div className="flex items-start justify-between gap-3">
   488→                  <dt className="text-muted-foreground">{tr.reference}</dt>
   489→                  <dd className="min-w-0 text-end">
   490→                    <a
   491→                      href={detail.referenceUrl}
   492→                      target="_blank"
   493→                      rel="noopener noreferrer"
   494→                      className="block truncate text-brand transition-colors hover:text-brand-strong ltr-isolate"
   495→                    >
   496→                      {detail.referenceUrl}
   497→                    </a>
   498→                  </dd>
   499→                </div>
   500→              ) : null}
   501→              <div className="flex items-start justify-between gap-3">
   502→                <dt className="text-muted-foreground">{t.account.profile.language}</dt>
   503→                <dd>{localeMeta[detail.locale === "en" ? "en" : "ar"].label}</dd>
   504→              </div>
   505→              <div className="flex items-start justify-between gap-3">
   506→                <dt className="text-muted-foreground">{t.admin.users.createdAt}</dt>
   507→                <dd className="text-end">{fmtDate(detail.createdAt, locale)}</dd>
   508→              </div>
   509→            </dl>
   510→          </section>
   511→
   512→          {/* الإدارة: تعيين/أولوية/حالة/أرشفة */}
   513→          <section className="space-y-4 rounded-2xl border border-border bg-white p-4">
   514→            {mayAssign ? (
   515→              <div className="space-y-2">
   516→                <Label>{tr.filterAssignee}</Label>
   517→                <div className="flex items-center gap-2">
   518→                  <Select
   519→                    value={detail.assigneeId ?? "none"}
   520→                    onValueChange={(v) => patch({ assigneeId: v === "none" ? null : v }, v === "none" ? tr.unassigned : undefined)}
   521→                    disabled={busy}
   522→                  >
   523→                    <SelectTrigger className="min-h-11 flex-1">
   524→                      <SelectValue />
   525→                    </SelectTrigger>
   526→                    <SelectContent>
   527→                      <SelectItem value="none">{tr.unassigned}</SelectItem>
   528→                      {staffOptions.map((s) => (
   529→                        <SelectItem key={s.id} value={s.id}>
   530→                          {s.name}
   531→                          {s.id === me.id ? ` (${tr.assignToMe})` : ""}
   532→                        </SelectItem>
   533→                      ))}
   534→                    </SelectContent>
   535→                  </Select>
   536→                  <Button
   537→                    variant="outline"
   538→                    size="icon"
   539→                    className="size-11 shrink-0"
   540→                    disabled={busy || detail.assigneeId === me.id}
   541→                    onClick={() => patch({ assigneeId: me.id }, me.name)}
   542→                    aria-label={tr.assignToMe}
   543→                    title={tr.assignToMe}
   544→                  >
   545→                    <UserPlus className="size-4" aria-hidden="true" />
   546→                  </Button>
   547→                </div>
   548→              </div>
   549→            ) : null}
   550→
   551→            {mayStatus ? (
   552→              <div className="space-y-2">
   553→                <Label>{tr.priority}</Label>
   554→                <Select
   555→                  value={detail.priority}
   556→                  onValueChange={(v) => patch({ priority: v }, tr.priorities[v] ?? v)}
   557→                  disabled={busy}
   558→                >
   559→                  <SelectTrigger className="min-h-11">
   560→                    <SelectValue />
   561→                  </SelectTrigger>
   562→                  <SelectContent>
   563→                    {PRIORITY_KEYS.map((p) => (
   564→                      <SelectItem key={p} value={p}>{tr.priorities[p]}</SelectItem>
   565→                    ))}
   566→                  </SelectContent>
   567→                </Select>
   568→              </div>
   569→            ) : null}
   570→
   571→            {mayStatus ? (
   572→              <div className="space-y-2">
   573→                <Label>{tr.changeStatus}</Label>
   574→                <Select value={statusTarget || detail.status} onValueChange={setStatusTarget}>
   575→                  <SelectTrigger className="min-h-11">
   576→                    <SelectValue />
   577→                  </SelectTrigger>
   578→                  <SelectContent>
   579→                    {STATUS_KEYS.map((s) => (
   580→                      <SelectItem key={s} value={s}>{tr.statuses[s]}</SelectItem>
   581→                    ))}
   582→                  </SelectContent>
   583→                </Select>
   584→                {isClosureTarget ? (
   585→                  <div className="space-y-1.5">
   586→                    <Label htmlFor="close-reason" className="text-destructive">{tr.closeReason}</Label>
   587→                    <Input
   588→                      id="close-reason"
   589→                      value={closeReason}
   590→                      onChange={(e) => setCloseReason(e.target.value)}
   591→                      placeholder={tr.closeReasonPlaceholder}
   592→                      maxLength={500}
   593→                      className="min-h-11"
   594→                    />
   595→                  </div>
   596→                ) : (
   597→                  <Input
   598→                    value={statusNote}
   599→                    onChange={(e) => setStatusNote(e.target.value)}
   600→                    placeholder={tr.statusNotePlaceholder}
   601→                    maxLength={500}
   602→                    className="min-h-11"
   603→                    aria-label={tr.changeStatus}
   604→                  />
   605→                )}
   606→                <Button
   607→                  onClick={applyStatus}
   608→                  disabled={applying || busy || (statusTarget || detail.status) === detail.status}
   609→                  className="min-h-11 w-full rounded-full"
   610→                >
   611→                  {applying ? <Loader2 className="size-4 animate-spin" aria-hidden="true" /> : null}
   612→                  {tr.applyStatus}
   613→                </Button>
   614→              </div>
   615→            ) : null}
   616→
   617→            {mayArchive ? (
   618→              <Button
   619→                variant={detail.archivedAt ? "outline" : "destructive"}
   620→                onClick={() =>
   621→                  patch({ action: detail.archivedAt ? "restore" : "archive" }, detail.archivedAt ? tr.restore : tr.archived)
   622→                }
   623→                disabled={busy}
   624→                className="min-h-11 w-full rounded-full"
   625→              >
   626→                {detail.archivedAt ? (
   627→                  <ArchiveRestore className="size-4" aria-hidden="true" />
   628→                ) : (
   629→                  <Archive className="size-4" aria-hidden="true" />
   630→                )}
   631→                {detail.archivedAt ? tr.restore : tr.archive}
   632→              </Button>
   633→            ) : null}
   634→          </section>
   635→
   636→          {/* المرفقات */}
   637→          <section className="rounded-2xl border border-border bg-white p-4">
   638→            <h2 className="text-sm font-semibold text-navy">{td.attachments}</h2>
   639→            {detail.attachments.length === 0 ? (
   640→              <p className="mt-2 text-xs text-muted-foreground">{tr.none}</p>
   641→            ) : (
   642→              <ul className="mt-3 space-y-1.5">
   643→                {detail.attachments.map((file: AttachmentRow) => (
   644→                  <li key={file.id}>
   645→                    <a
   646→                      href={`/api/attachments/${file.id}`}
   647→                      className="flex items-center gap-2 rounded-xl border border-border px-3 py-2 text-sm transition-colors hover:border-brand hover:bg-accent/40"
   648→                    >
   649→                      <Paperclip className="size-3.5 shrink-0 text-brand" aria-hidden="true" />
   650→                      <span className="min-w-0 flex-1 truncate ltr-isolate">{file.filename}</span>
   651→                      <span className="shrink-0 text-xs tabular-nums text-muted-foreground">{formatBytes(file.size)}</span>
   652→                      <Download className="size-3.5 shrink-0 text-muted-foreground" aria-hidden="true" />
   653→                    </a>
   654→                  </li>
   655→                ))}
   656→              </ul>
   657→            )}
   658→          </section>
   659→
   660→          {/* الخط الزمني للحالة */}
   661→          <section className="rounded-2xl border border-border bg-white p-4">
   662→            <h2 className="flex items-center gap-2 text-sm font-semibold text-navy">
   663→              <History className="size-4 text-brand" aria-hidden="true" />
   664→              {td.statusTimeline}
   665→            </h2>
   666→            {detail.statusHistory.length === 0 ? (
   667→              <p className="mt-2 text-xs text-muted-foreground">{tr.none}</p>
   668→            ) : (
   669→              <ol className="mt-3 space-y-0">
   670→                {detail.statusHistory.map((event) => (
   671→                  <li key={event.id} className="relative border-s-2 border-border ps-4 pb-4 last:pb-0">
   672→                    <span
   673→                      className={cn(
   674→                        "absolute -start-[5px] top-1 size-2 rounded-full",
   675→                        event.toStatus === "closed" || event.toStatus === "cancelled" ? "bg-muted-foreground" : "bg-brand"
   676→                      )}
   677→                      aria-hidden="true"
   678→                    />
   679→                    <p className="text-xs font-semibold text-navy">
   680→                      {event.fromStatus ? `${tr.statuses[event.fromStatus] ?? event.fromStatus} → ` : ""}
   681→                      {tr.statuses[event.toStatus] ?? event.toStatus}
   682→                    </p>
   683→                    <p className="mt-0.5 text-[11px] text-muted-foreground">
   684→                      {event.changedBy?.name ?? "—"} · {fmtDateTime(event.createdAt, locale)}
   685→                    </p>
   686→                    {event.note ? (
   687→                      <p className="mt-1 rounded-lg bg-muted px-2 py-1 text-xs text-muted-foreground">{event.note}</p>
   688→                    ) : null}
   689→                  </li>
   690→                ))}
   691→              </ol>
   692→            )}
   693→            {detail.resolutionNote ? (
   694→              <p className="mt-2 rounded-xl bg-muted px-3 py-2 text-xs text-muted-foreground">
   695→                {td.closedNote}: {detail.resolutionNote}
   696→              </p>
   697→            ) : null}
   698→          </section>
   699→        </aside>
   700→      </div>
   701→    </div>
   702→  );
